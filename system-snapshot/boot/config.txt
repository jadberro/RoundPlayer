# --- Audio: Raspberry Pi DAC Pro (PCM5242) is the only sound card ----------
# Kill the on-board headphone PWM DAC and HDMI audio so PipeWire only sees the DAC Pro.
dtparam=audio=off
dtoverlay=vc4-kms-v3d,noaudio
dtoverlay=rpi-dacpro

# --- Display: 5" round 1080x1080 HDMI LCD (Waveshare sku 24369) -----------
# Under KMS the panel's EDID normally advertises 1080x1080@60 and nothing else is needed.
# If the Pi comes up at the wrong mode, force it from the kernel command line instead of
# legacy hdmi_* keys (those are ignored by vc4-kms-v3d):
#   sudo sed -i 's/$/ video=HDMI-A-1:1080x1080M@60D/' /boot/firmware/cmdline.txt
# Keep the HDMI link powered even if the panel is slow to answer EDID.
hdmi_force_hotplug=1
max_framebuffers=2
disable_overscan=1

# --- GPU / decode ----------------------------------------------------------
# 1080p H.264 is decoded by the V4L2 stateful decoder (bcm2835-codec); mpv uses hwdec=v4l2m2m-copy.
gpu_mem=256
arm_boost=1
